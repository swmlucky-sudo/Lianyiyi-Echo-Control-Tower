async function compile() { const input = document.getElementById('user-input').value.trim(); if(!input) return; document.getElementById('log-display').innerHTML = ""; document.getElementById('core').classList.remove('locked'); log("INTENT", `分析橋主意圖語義特徵...`, "#fff"); // --- 精密權重算法核心 --- let base = { en: 20, jp: 20, cn: 20, tw: 20 }; // 1. 邏輯特徵掃描 (EN) const techMarkers = (input.match(/[a-zA-Z0-9{}<>[\]();=]/g) || []).length; base.en += Math.min(techMarkers * 5, 75); // 符號與代碼越多，EN 越高 // 2. 情感特徵掃描 (JP) const emotionalMarkers = (input.match(/[!！?？...。，累煩煩憂傷心好]/g) || []).length; base.jp += Math.min(emotionalMarkers * 8, 75); // 3. 敘事長度掃描 (CN) base.cn += Math.min(input.length * 2, 75); // 找出主導聲部 let winner = 'cn'; if (base.en >= base.jp && base.en >= base.cn) winner = 'en'; else if (base.jp >= base.en && base.jp >= base.cn) winner = 'jp'; // 4. 台語收斂係數 (基於主導聲部的信心值) base.tw = Math.min(base[winner] + 10, 98); // --- 視覺渲染過程 --- // 階段一：分流 (反映原始掃描數據) log("PROCESS", "執行特徵向量提取..."); setBar('en', Math.min(base.en, 95)); setBar('jp', Math.min(base.jp, 95)); setBar('cn', Math.min(base.cn, 95)); setBar('tw', 30); if(winner === 'en') log("EN_DATA", `>> 偵測到代碼/邏輯標記 ${techMarkers} 處，優先處理。`, "#58a6ff"); if(winner === 'jp') log("JP_DATA", `>> 偵測到情感/語氣偏移 ${emotionalMarkers} 處，啟動緩衝。`, "#ff7b72"); if(winner === 'cn') log("CN_DATA", `>> 基於文字長度 ${input.length} 展開脈絡分析。`, "#7ee787"); await new Promise(r => setTimeout(r, 1200)); // 階段二：台語定音 (依據主導信心進行最後裁斷) log("KERNEL", `台語聲部介入：依據「${winner.toUpperCase()}」信心值執行定音。`, "#d29922"); setBar('en', 15); setBar('jp', 15); setBar('cn', 15); setBar('tw', base.tw); await new Promise(r => setTimeout(r, 1000)); // 階段三：鎖定 document.getElementById('core').classList.add('locked'); document.getElementById('decision-text').innerText = `系統定音完成。信心值: ${base.tw}%`; log("SUCCESS", `440Hz 穩態鎖定。`, "#7ee787"); setBar('en', 35); setBar('jp', 35); setBar('cn', 35); setBar('tw', 35); }
